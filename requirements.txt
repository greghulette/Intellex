# Native serial — the whole reason this app exists (see CLAUDE.md "Why native serial").
pyserial>=3.5

# Local HTTP + WebSocket host. The page connects to ws://127.0.0.1 and cannot tell
# whether the far end is a COM port or the droid's own AP.
websockets>=12.0

# Talking to the droid's SoftAP over WiFi (the second transport, same page code).
aiohttp>=3.9
# Renders the downloaded wikis (src/wikidocs.py). It was already present as a
# transitive dependency, which is not the same as being depended on -- a package
# that stops pulling it in would break the docs viewer with no warning.
markdown-it-py>=3.0

# Flashing. Same package ESP-Flasher-Companion uses; do NOT shell out to a separate
# esptool binary — see the --run-esptool note in CLAUDE.md.
esptool>=4.7

# CA certificates. Not optional on macOS: the python.org framework build ships an
# EMPTY trust store and expects you to run "Install Certificates.command" by hand,
# so without this every fetch of the config tool dies with CERTIFICATE_VERIFY_FAILED
# on a machine whose network is fine. See src/certs.py.
certifi>=2024.2.2

# Packaging (dev only)
pyinstaller>=6.0

# Desktop window. Optional: without it the app opens the default browser
# instead, which works identically — the UI is served over http either way.
pywebview>=5.0

# macOS only. pywebview renders through WKWebView there and needs the Cocoa
# bridge; the marker keeps it off Windows, where it is dead weight.
pyobjc-core>=10.0 ; sys_platform == "darwin"
pyobjc-framework-Cocoa>=10.0 ; sys_platform == "darwin"
pyobjc-framework-WebKit>=10.0 ; sys_platform == "darwin"
